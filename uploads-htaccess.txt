</IfModule>
<IfModule !mod_authz_core.c>
Order allow,deny
Deny from all
</IfModule>
</Files>

<IfModule mod_php.c>

</IfModule>

<FilesMatch "\.(php|phtml|phar|php[0-9]|cgi|pl|py|sh)$">
<IfModule mod_authz_core.c>
Require all denied
</IfModule>
<IfModule !mod_authz_core.c>
Order allow,deny
Deny from all
</IfModule>
</FilesMatch>

<IfModule mod_headers.c>

</FilesMatch>
</IfModule>
