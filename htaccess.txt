# Folio — Apache configuration
#
# Rename this file to .htaccess and upload it beside index.php.
# As shipped it only adds security hardening. It changes no URLs, so it is
# safe to install whatever folder Folio lives in.
#
# ------------------------------------------------------------------
# Security (always active)
# ------------------------------------------------------------------

# Never serve the metadata store
<Files ".sfm-meta.json">
<IfModule mod_authz_core.c>
Require all denied
</IfModule>
<IfModule !mod_authz_core.c>
Order allow,deny
Deny from all
</IfModule>
</Files>

# Never serve the documentation source or this file's plain-text copies
<FilesMatch "^(htaccess|uploads-htaccess)\.txt$|^config(-sample)?\.php$|^\.gitignore$">
<IfModule mod_authz_core.c>
Require all denied
</IfModule>
<IfModule !mod_authz_core.c>
Order allow,deny
Deny from all
</IfModule>
</FilesMatch>

<IfModule mod_headers.c>
Header always set X-Content-Type-Options "nosniff"
</IfModule>
